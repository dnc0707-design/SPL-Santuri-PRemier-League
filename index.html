Master Admin Dashboard

Admin access is controlled by Firebase Authentication and the server-assigned admin: true custom claim.

Firebase Database Rules Code

In Firebase Console → Firestore Database → Rules, publish this secure ruleset. Admin access also requires a server-assigned Firebase custom claim: admin: true. Do not use an open read/write rule.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isAdmin() { return signedIn() && request.auth.token.admin == true; }
    function isAnonymous() {
      return signedIn() && request.auth.token.firebase.sign_in_provider == 'anonymous';
    }

    // Public website content is readable by visitors; only admins can manage it.
    match /artifacts/{appId}/public/data/{collectionName}/{documentId} {
      allow read: if isAdmin() || (collectionName != 'pollVotes' && collectionName != 'players');
      allow write: if isAdmin() && collectionName != 'pollVotes';
    }

    // Player applications are private. Only their initial submission is public-facing.
    match /artifacts/{appId}/private/data/registrations/{registrationId} {
      allow create: if (isAnonymous() || isAdmin())
        && request.resource.data.keys().hasOnly([
          'isRegistration', 'name', 'dob', 'panchayat', 'village', 'phone',
          'skill', 'photo', 'aadhaar', 'payment', 'status', 'srNo', 'createdAt'
        ])
        && request.resource.data.isRegistration == true
        && request.resource.data.status == 'pending'
        && request.resource.data.name is string
        && request.resource.data.name.size() > 1
        && request.resource.data.name.size() <= 80
        && request.resource.data.phone is string
        && request.resource.data.phone.matches('^[0-9]{10}$')
        && request.resource.data.photo is string
        && request.resource.data.photo.size() < 300000
        && request.resource.data.aadhaar is string
        && request.resource.data.aadhaar.size() < 300000
        && request.resource.data.payment is string
        && request.resource.data.payment.size() < 300000;
      allow read, update, delete: if isAdmin();
    }

    // Franchise-owner applications remain private until reviewed by an admin.
    match /artifacts/{appId}/private/data/owner_registrations/{registrationId} {
      allow create: if (isAnonymous() || isAdmin())
        && request.resource.data.keys().hasOnly([
          'isOwnerRegistration', 'franchiseName', 'owners', 'phone', 'address',
          'payment', 'declaration', 'status', 'franchiseRegId', 'createdAt'
        ])
        && request.resource.data.isOwnerRegistration == true
        && request.resource.data.status == 'pending'
        && request.resource.data.franchiseName is string
        && request.resource.data.franchiseName.size() > 1
        && request.resource.data.franchiseName.size() <= 80
        && request.resource.data.owners is list
        && request.resource.data.owners.size() > 0
        && request.resource.data.owners.size() <= 6
        && request.resource.data.phone is string
        && request.resource.data.phone.matches('^[0-9]{10}$')
        && request.resource.data.address is string
        && request.resource.data.address.size() <= 300
        && request.resource.data.payment is string
        && request.resource.data.payment.size() < 300000
        && request.resource.data.declaration == true;
      allow read, update, delete: if isAdmin();
    }

    // A visitor can create one vote marker per poll, only for an existing option.
    match /artifacts/{appId}/public/data/pollVotes/{voteId} {
      allow create: if isAnonymous()
        && !exists(/databases/$(database)/documents/artifacts/$(appId)/public/data/pollVotes/$(voteId))
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.pollId is string
        && voteId == request.resource.data.pollId + '_' + request.auth.uid
        && request.resource.data.optionIndex is int
        && request.resource.data.optionIndex >= 0
        && request.resource.data.optionIndex <= 3
        && exists(/databases/$(database)/documents/artifacts/$(appId)/public/data/polls/$(request.resource.data.pollId))
        && (
          (request.resource.data.optionIndex == 0 && get(/databases/$(database)/documents/artifacts/$(appId)/public/data/polls/$(request.resource.data.pollId)).data.opt0 is string && get(/databases/$(database)/documents/artifacts/$(appId)/public/data/polls/$(request.resource.data.pollId)).data.opt0.size() > 0)
          || (request.resource.data.optionIndex == 1 && get(/databases/$(database)/documents/artifacts/$(appId)/public/data/polls/$(request.resource.data.pollId)).data.opt1 is string && get(/databases/$(database)/documents/artifacts/$(appId)/public/data/polls/$(request.resource.data.pollId)).data.opt1.size() > 0)
          || (request.resource.data.optionIndex == 2 && get(/databases/$(database)/documents/artifacts/$(appId)/public/data/polls/$(request.resource.data.pollId)).data.opt2 is string && get(/databases/$(database)/documents/artifacts/$(appId)/public/data/polls/$(request.resource.data.pollId)).data.opt2.size() > 0)
          || (request.resource.data.optionIndex == 3 && get(/databases/$(database)/documents/artifacts/$(appId)/public/data/polls/$(request.resource.data.pollId)).data.opt3 is string && get(/databases/$(database)/documents/artifacts/$(appId)/public/data/polls/$(request.resource.data.pollId)).data.opt3.size() > 0)
        );
      allow read, update, delete: if isAdmin();
    }

    // Public poll reads are allowed; visitors can only increment the option they marked.
    match /artifacts/{appId}/public/data/polls/{pollId} {
      allow read: if true;
      allow create, delete: if isAdmin();
      allow update: if isAdmin() || (
        isAnonymous()
        && existsAfter(/databases/$(database)/documents/artifacts/$(appId)/public/data/pollVotes/$(pollId + '_' + request.auth.uid))
        && !exists(/databases/$(database)/documents/artifacts/$(appId)/public/data/pollVotes/$(pollId + '_' + request.auth.uid))
        && getAfter(/databases/$(database)/documents/artifacts/$(appId)/public/data/pollVotes/$(pollId + '_' + request.auth.uid)).data.uid == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['val0', 'val1', 'val2', 'val3'])
        && (
          (getAfter(/databases/$(database)/documents/artifacts/$(appId)/public/data/pollVotes/$(pollId + '_' + request.auth.uid)).data.optionIndex == 0
            && resource.data.opt0 is string && resource.data.opt0.size() > 0
            && request.resource.data.val0 == resource.data.val0 + 1
            && request.resource.data.val1 == resource.data.val1 && request.resource.data.val2 == resource.data.val2 && request.resource.data.val3 == resource.data.val3)
          || (getAfter(/databases/$(database)/documents/artifacts/$(appId)/public/data/pollVotes/$(pollId + '_' + request.auth.uid)).data.optionIndex == 1
            && resource.data.opt1 is string && resource.data.opt1.size() > 0
            && request.resource.data.val1 == resource.data.val1 + 1
            && request.resource.data.val0 == resource.data.val0 && request.resource.data.val2 == resource.data.val2 && request.resource.data.val3 == resource.data.val3)
          || (getAfter(/databases/$(database)/documents/artifacts/$(appId)/public/data/pollVotes/$(pollId + '_' + request.auth.uid)).data.optionIndex == 2
            && resource.data.opt2 is string && resource.data.opt2.size() > 0
            && request.resource.data.val2 == resource.data.val2 + 1
            && request.resource.data.val0 == resource.data.val0 && request.resource.data.val1 == resource.data.val1 && request.resource.data.val3 == resource.data.val3)
          || (getAfter(/databases/$(database)/documents/artifacts/$(appId)/public/data/pollVotes/$(pollId + '_' + request.auth.uid)).data.optionIndex == 3
            && resource.data.opt3 is string && resource.data.opt3.size() > 0
            && request.resource.data.val3 == resource.data.val3 + 1
            && request.resource.data.val0 == resource.data.val0 && request.resource.data.val1 == resource.data.val1 && request.resource.data.val2 == resource.data.val2)
        )
      );
    }
  }
}
